acumularMillas: {identificador_usuario (Z), distancia_volada (Z), nombre_promocion (string)}
- agrega millas a un usuario, actualiza la categoria si corresponde
- se registra la transaccion en el historial del usuario (distancia volada, millas ganadas, nombre y factor de promocion)
- controla el tope de la promocion (actualiza, y si es necesario elemina)

canjearMillas: {identificador_usuario (Z), millas_gastadas (Z)}
- descuenta millas disponibles del usuario
- se registra la transaccion en el historial (cantidad de millas canjeadas)

transferirMillas: {identificador_usuario (Z), millas_transferidas (Z), identificador_referente (Z)}
- transfiere las millas de un usuario a otro
- se registra la transaccion en el historial del usuario (millas transferidas, usuario destino, usuario origen)

consultarSaldo: {identificador_usuario (Z)} --> Z
- devuelve las millas disponibles del usuario

consultarCategoria: {identificador_usuario (Z)} --> string
- devuelve la categoria del usuario

rankingMillas: {top_cuantos (Z)} --> seq<Z>
- devuelve los n usuarios con mas millas disponibles, de mayor a menor
- devuelve todos si n es mayor a la cantidad de usuarios totales

resumenTransacciones: {identificador_usuario (Z)} --> tupla<Z,Z,Z>
- devuelve las millas ganadas por viajes, canjeadas, millas que transfirio y millas que le fueron transferidas

iniciarPromocion: (inout promociones : conj<struc<PROMO>> , in nombre_promocion: String, in factor: R)
{
    requiere {noEsta(promociones, nombre_promocion)}
    requiere {factor >= 1}
    requiere {promociones = PROMOCIONES0}
    asegura {}
}

pred noEsta (promociones : conj<struc<PROMO>>, nueva_promo : String)
{
    (Vp : struct<PROMO>) p.nombre_promocion != nueva_promo
}


iniciarPromocionConTope {nombre_promocion (string), factor (R), tope (Z)}
- agrega una promocion activa
- el tope indica hasta cuantas millas se pueden ganar con ella

proc finalizarPromocion (inout promocion : info_promo  , in nombre_promocion: String)
{
    requiere {nombre_promocion != "NoPromo"}
    requiere {nombre_promocion E promocion.claves}
    requiere {promocion = PROMOCIONES0}

    asegura  {promocion = delKey(PROMOCIONES0, nombre_promocion)}
}
